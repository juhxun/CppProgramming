```cpp
#include <iostream>
- iostream 라이브러리 포함
#include <string>
- string 클래스 사용
using namespace std;
- std 네임스페이스 사용
class WordUtility {
- 문자열 단어 처리를 위한 클래스
public:
- 외부에서 접근 가능한 멤버
    static int wordCount(string s) {
- 문자열의 단어 개수를 세는 static 함수
        int count = 0;
- 공백 개수 저장 변수
        for (int i = 0; i < s.length(); i++) {
- 문자열 전체 반복
            if (s[i] == ' ')
- 공백 문자 확인
                count++;
- 공백 개수 증가
        }
        return count + 1;
- 단어 수 반환
- 단어 개수 = 공백 개수 + 1
    }
    static string getWord(string s, int n) {
- n번째 단어를 반환하는 static 함수
        int count = 1;
- 현재 단어 번호
        string word = "";
- 단어 저장 변수
        for (int i = 0; i <= s.length(); i++) {
- 문자열 전체 반복
- 마지막 문자까지 검사하기 위해 <= 사용
            if (s[i] == ' ' || s[i] == '\0') {
- 공백 또는 문자열 끝 확인
                if (count == n)
- 현재 단어가 찾는 단어인지 확인
                    return word;
- n번째 단어 반환
                word = "";
- 다음 단어를 위해 초기화
                count++;
- 단어 번호 증가
            }
            else {
- 일반 문자일 경우
                word += s[i];
- 현재 단어에 문자 추가
            }
        }
        return "";
- 해당 단어가 없으면 빈 문자열 반환
    }
};
int main() {
- 프로그램 시작점
    int n = WordUtility::wordCount("I love C++");
- 단어 개수 계산
    cout << "단어 개수는 " << n << endl;
- 단어 개수 출력
    string word = WordUtility::getWord("I love C++", 3);
- 3번째 단어 가져오기
    if (word == "")
- 단어가 없는지 확인
        cout << "3번째 단어는 없습니다." << endl;
- 없으면 출력
    else
- 단어가 존재하면
        cout << "3번째 단어는 " << word << endl;
- 단어 출력
    return 0;
- 프로그램 종료
}
