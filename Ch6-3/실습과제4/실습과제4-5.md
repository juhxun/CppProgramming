# 실습과제4-5.md



```cpp
#include <iostream>
- 입출력을 위한 iostream 라이브러리 포함

using namespace std;
- std 네임스페이스 사용

class Vector {
- 정수 데이터를 저장하는 벡터(Vector) 클래스

    int* mem;
- 동적 배열을 가리키는 포인터

    int size;
- 배열 크기 저장 변수

public:
- 외부에서 접근 가능한 멤버

    Vector() {
- 기본 생성자

        size = 5;
- 배열 크기를 5로 설정

        mem = new int[size];
- 크기 5인 동적 배열 생성

        for (int i = 0; i < size; i++)
- 배열 전체 반복

            mem[i] = 0;
- 모든 값을 0으로 초기화

    }

    Vector(int s, int value) {
- 매개변수 생성자
- 배열 크기와 초기값 설정

        size = s;
- 배열 크기 저장

        mem = new int[size];
- 동적 배열 생성

        for (int i = 0; i < size; i++)
- 배열 전체 반복

            mem[i] = value;
- 모든 값을 value로 초기화

    }

    ~Vector() {
- 소멸자

        delete[] mem;
- 동적 메모리 해제

    }

    void show() {
- 배열 전체 출력 함수

        for (int i = 0; i < size; i++)
- 배열 전체 반복

            cout << mem[i] << " ";
- 배열 값 출력

        cout << endl;
- 줄바꿈

    }

    void show(int n) {
- 배열 앞 n개만 출력하는 함수
- 함수 오버로딩 사용

        for (int i = 0; i < n; i++)
- n번 반복

            cout << mem[i] << " ";
- 값 출력

        cout << endl;
- 줄바꿈

    }
};

int main() {
- 프로그램 시작점

    Vector x;
- 기본 생성자 호출
- 크기 5, 값은 모두 0

    Vector y(10, 8);
- 크기 10, 값은 모두 8인 객체 생성

    x.show();
- x 전체 출력

    y.show();
- y 전체 출력

    y.show(3);
- y의 앞 3개 값만 출력

    return 0;
- 프로그램 종료
}
