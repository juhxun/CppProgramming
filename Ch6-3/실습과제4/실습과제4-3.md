#include <iostream>
- iostream 라이브러리 포함
#include <string>
- string 클래스 사용
using namespace std;
- std 네임스페이스 사용
bool equals(string x, string y) {
- 문자열 전체 비교 함수
    return x == y;
- 두 문자열이 같으면 true 반환
}
bool equals(string x, string y, int n) {
- 문자열 앞 n글자 비교 함수
- 함수 오버로딩 사용
    return x.substr(0, n) == y.substr(0, n);
- substr(0,n) → 0번 인덱스부터 n글자 추출
- 앞 n글자가 같으면 true 반환
}
bool equals(string x, string y, int n, string z) {
- 매개변수 4개 버전 함수
- x와 z의 앞 n글자 비교
    return x.substr(0, n) == z.substr(0, n);
- 앞 n글자 비교 결과 반환
}
int main() {
- 프로그램 시작점
    string x = "Prof. Hwang";
- 문자열 x 초기화
    string y = "Prof. Kim";
- 문자열 y 초기화
    string z = "Prof. Lee";
- 문자열 z 초기화
    if (equals(x, y))
- 문자열 전체 비교
        cout << "같음" << endl;
- 같으면 출력
    if (equals(x, y, 3))
- x와 y의 앞 3글자 비교
        cout << "앞 3글자 같음" << endl;
- 같으면 출력
    if (equals(x, y, 5, z))
- x와 z의 앞 5글자 비교
        cout << "앞 5글자 같음" << endl;
- 같으면 출력
    return 0;
- 프로그램 종료
}
