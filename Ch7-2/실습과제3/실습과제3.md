#include <iostream>
- 입출력을 위한 iostream 라이브러리 포함
using namespace std;
- std 네임스페이스 사용
class Complex {
- 복소수 객체를 정의하는 클래스
private:
- 클래스 내부에서만 접근 가능
    int real;
- 복소수의 실수부 저장
    int img;
- 복소수의 허수부 저장
public:
- 외부에서 접근 가능한 멤버
    Complex(int r = 0, int i = 0) {
- 생성자 정의
        real = r;
- 실수부 초기화
        img = i;
- 허수부 초기화
    }
    Complex& operator+=(const Complex& c) {
- += 연산자 오버로딩
        real += c.real;
- 실수부 더하기
        img += c.img;
- 허수부 더하기
        return *this;
- 현재 객체 자신 반환
    }
    void show() {
- 복소수 출력 함수
        cout << real << "+" << img << "i" << endl;
- 복소수 형태 출력
    }
};
int main() {
- 프로그램 시작점
    Complex a(3, 5), b(1, 1);
- 복소수 객체 생성
    a.show();
- a 출력
    b.show();
- b 출력
    b += a;
- 오버로딩된 += 연산자 호출
    a.show();
- a 출력
- 값은 변경되지 않음
    b.show();
- 변경된 b 출력
    return 0;
- 프로그램 종료
}
