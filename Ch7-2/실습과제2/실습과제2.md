# 실습과제2



```cpp
#include <iostream>
- iostream 라이브러리 포함
using namespace std;
- std 네임스페이스 사용
class Complex {
- 복소수 객체를 정의하는 클래스
private:
- 클래스 내부에서만 접근 가능
    int real;
- 복소수의 실수부 저장
    int img;
- 복소수의 허수부 저장
public:
- 외부에서 접근 가능한 멤버
    Complex(int r = 0, int i = 0) {
- 생성자 정의
        real = r;
- 실수부 초기화
        img = i;
- 허수부 초기화
    }
    Complex operator*(const Complex& c) {
- 곱셈 연산자(*) 오버로딩
        Complex temp;
- 결과 저장용 객체 생성
        temp.real = real * c.real - img * c.img;
- 복소수 곱셈 공식의 실수부 계산
        temp.img = real * c.img + img * c.real;
- 복소수 곱셈 공식의 허수부 계산
        return temp;
- 결과 객체 반환
    }
    void show() {
- 복소수 출력 함수
        cout << real;
- 실수부 출력
        if (img >= 0)
- 허수부가 양수이면
            cout << "+" << img << "i" << endl;
- +기호 포함 출력
        else
- 허수부가 음수이면
            cout << img << "i" << endl;
- 음수 기호 그대로 출력
    }
};
int main() {
- 프로그램 시작점
    Complex x(2, 3), y(-5, 10), sum;
- 복소수 객체 생성
    sum = x * y;
- 오버로딩된 * 연산자 호출
    cout << "두 복소수의 곱은 ";
    sum.show();
- 계산 결과 출력
    return 0;
- 프로그램 종료
}
