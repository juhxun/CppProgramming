#include <iostream>
- 입출력을 위한 iostream 라이브러리 포함
using namespace std;
- std 네임스페이스 사용
class Complex {
- 복소수 객체를 정의하는 클래스
    int real;
- 실수부 저장 변수
    int img;
- 허수부 저장 변수
public:
- 외부에서 접근 가능
    Complex(int r = 0, int i = 0) {
- 생성자 정의
        real = r;
- 실수부 초기화
        img = i;
- 허수부 초기화
    }
    void show() {
- 복소수 출력 함수
        cout << real << "+" << img << "j" << endl;
- 복소수 출력
    }
    friend Complex& operator++(Complex& c);
    friend Complex operator--(Complex& c, int);
};
Complex& operator++(Complex& c) {
- 전위 증가 연산자 함수 정의
    c.real++;
- 실수부 1 증가
    c.img++;
- 허수부 1 증가
    return c;
- 증가된 객체 반환
}
Complex operator--(Complex& c, int) {
- 후위 감소 연산자 함수 정의
    Complex temp = c;
- 감소 전 상태 저장
    c.real--;
- 실수부 1 감소
    c.img--;
- 허수부 1 감소
    return temp;
- 감소 전 객체 반환
}
int main() {
- 프로그램 시작점
    Complex x(2, -3);
- x = 2-3j 객체 생성
    ++x;
- 전위 증가 연산 실행
    cout << "증가결과 ";
    x.show();
- 증가 후 결과 출력
    x--;
- 후위 감소 연산 실행
    cout << "감소결과 ";
    x.show();
- 감소 후 결과 출력
    return 0;
- 프로그램 종료
}
