#include <iostream>
- iostream 라이브러리 포함
using namespace std;
- std 네임스페이스 사용
class Power {
	int kick;
	int punch;
public:
- 외부에서 접근 가능
	Power(int kick = 0, int punch = 0) {
- 생성자
		this->kick = kick;
- 멤버 변수 kick 초기화
		this->punch = punch;
- 멤버 변수 punch 초기화
	}
	void show();
- 함수 선언
	Power operator*(int op2);
- 객체 * 정수 연산자 오버로딩 선언
	Power operator+(Power op2);
- 객체 + 객체 연산자 오버로딩 선언
	friend Power operator*(int op1, Power op2);
- 정수 * 객체 연산자 오버로딩 선언
};
void Power::show() {
- 함수 정의
	cout << "kick=" << kick << ',' << "punch=" << punch << endl;
- kick와 punch 값 출력
}
Power Power::operator*(int op2) {
- 객체 * 정수 연산자 오버로딩
	Power tmp;
- 결과 저장 객체 생성
	tmp.kick = this->kick * op2;
- kick 값에 op2 곱하기
	tmp.punch = this->punch * op2;
- punch 값에 op2 곱하기
	return tmp;
- 결과 반환
}
Power Power::operator+(Power op2) {
- 객체 + 객체 연산자 오버로딩
	Power c;
- 결과 저장 객체 생성
	c.kick = this->kick + op2.kick;
- kick 값끼리 더하기
	c.punch = this->punch + op2.punch;
- punch 값끼리 더하기
	return c;
- 결과 반환
}
Power operator*(int op1, Power op2) {
- 정수 * 객체 연산자 오버로딩
	Power tmp;
- 결과 저장 객체 생성
	tmp.kick = op1 * op2.kick;
- kick 값 계산
	tmp.punch = op1 * op2.punch;
- punch 값 계산
	return tmp;
- 결과 반환
}
int main() {
- 프로그램 시작점
	Power a(1, 1), b(2, 2), c;
- a = (1,1)
- b = (2,2)
- c = (0,0)
	c.show();
- c 출력
	c = a * 2 + 2 * b;
- 연산 수행
	c.show();
- 결과 출력
	return 0;
- 프로그램 종료
}
