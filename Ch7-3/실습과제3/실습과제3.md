#include <iostream>
- iostream 라이브러리 포함
using namespace std;
- std 네임스페이스 사용
class Power {
	int kick;
	int punch;
public:
- 외부에서 접근 가능한 멤버
	Power(int kick = 0, int punch = 0) {
- 생성자
		this->kick = kick;
- 멤버 변수 kick 초기화
		this->punch = punch;
- 멤버 변수 punch 초기화
	}
	void show();
- 함수 선언
	friend Power operator*(int op1, Power op2);
- 정수 × 객체 연산자 오버로딩
};
void Power::show() {
- 함수 정의
	cout << "kick=" << kick << ',' << "punch=" << punch << endl;
- kick, punch 값 출력
}
Power operator*(int op1, Power op2) {
- 정수 × 객체 연산자 오버로딩 함수 정의
	Power tmp;
- 결과 저장용 객체 생성
	tmp.kick = op1 * op2.kick;
- kick 값에 정수 곱하기
	tmp.punch = op1 * op2.punch;
- punch 값에 정수 곱하기
	return tmp;
- 결과 객체 반환
}
int main() {
- 프로그램 시작점
	Power a(3, 5), b;
- a = (3,5)
- b = (0,0)
	a.show();
- a 출력
	b.show();
- b 출력
	b = 2 * a;
- friend 연산자 함수 호출
- b = (6,10)
	a.show();
- a 출력
- 값은 변경되지 않음
	b.show();
- 변경된 b 출력
	return 0;
- 프로그램 종료
}
